-- ==============================================================================
-- 1. NODES TABLE (The Data Assets)
-- ==============================================================================
CREATE TABLE data_assets (
    asset_id VARCHAR(255) PRIMARY KEY,       -- e.g., 'catalog.schema.table'
    asset_name VARCHAR(255) NOT NULL,        -- e.g., 'bronze_sales'
    asset_type VARCHAR(50) NOT NULL,         -- 'table', 'view', 'dashboard', 'ml_model'
    status VARCHAR(50) DEFAULT 'SUCCESS',    -- 'SUCCESS', 'FAILED', 'RUNNING', 'STALE'
    last_updated_at TIMESTAMP,
    owner_email VARCHAR(255)                 -- Who to page if it breaks
);

-- ==============================================================================
-- 2. EDGES TABLE (The Lineage Connections)
-- ==============================================================================
CREATE TABLE data_lineage (
    upstream_asset_id VARCHAR(255) REFERENCES data_assets(asset_id),
    downstream_asset_id VARCHAR(255) REFERENCES data_assets(asset_id),
    PRIMARY KEY (upstream_asset_id, downstream_asset_id)
);
